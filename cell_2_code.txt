def load_model_cpu(model_id: str, use_quantization: bool = False):
    """Загрузка модели с оптимизацией для CPU"""
    print(f"\nЗагрузка модели: {model_id}")

        tokenizer = AutoTokenizer.from_pretrained(model_id, trust_remote_code=True)

            model_id,
            torch_dtype=torch.float32,
            low_cpu_mem_usage=True,
            trust_remote_code=True
        )
        
        model = model.to("cpu")
        
        pipe = pipeline(
            "text-generation",
            model=model,
            tokenizer=tokenizer,
            max_new_tokens=512,
            temperature=0.1,
            do_sample=False,
            repetition_penalty=1.1
        )
        
        print(f"  Модель успешно загружена")
        return pipe, model, tokenizer
        
    except Exception as e:
        print(f"  Ошибка загрузки модели: {e}")
        return None, None, None

class BatchProcessor:
    """Обработчик для пакетной обработки текстов"""
    
    def __init__(self, pipeline, batch_size: int = 4):
        self.pipeline = pipeline
        self.batch_size = batch_size
        self.cache = {}
        self.stats = {
            "total_processed": 0,
            "cache_hits": 0,
            "total_time": 0
        }
    
    def _create_cache_key(self, text: str) -> str:
        return hash(text) % 1000000
    
    def process_batch(self, texts: List[str]) -> List[Dict]:
        """Обработка пакета текстов"""
        results = []
        
        for i in range(0, len(texts), self.batch_size):
            batch = texts[i:i + self.batch_size]
            batch_results = []
            
            for text in batch:
                cache_key = self._create_cache_key(text)
                if cache_key in self.cache:
                    self.stats["cache_hits"] += 1
                    batch_results.append(self.cache[cache_key])
                else:
                    prompt = EXTRACTION_PROMPT.format(text=text)
                    result = self.pipeline(prompt)
                    
                    if isinstance(result, list) and len(result) > 0:
                        generated = result[0].get("generated_text", "")
                    else:
                        generated = str(result)
                    
                    try:
                        json_start = generated.find("{")
                        json_end = generated.rfind("}") + 1
                        if json_start >= 0 and json_end > json_start:
                            entities = json.loads(generated[json_start:json_end])
                        else:
                            entities = {"error": "JSON not found"}
                    except:
                        entities = {"error": "JSON parse failed"}
                    
                    self.cache[cache_key] = entities
                    batch_results.append(entities)
            
            results.extend(batch_results)
            self.stats["total_processed"] += len(batch)
        
        return results
    
    def get_stats(self) -> Dict:
        return self.stats

from concurrent.futures import ThreadPoolExecutor
import threading

class ParallelEntityExtractor:
    """Извлекатель сущностей с параллельной обработкой"""
    
    def __init__(self, pipeline, max_workers: int = 4):
        self.pipeline = pipeline
        self.max_workers = max_workers
        self.results = []
        self.lock = threading.Lock()
    
    def extract_single(self, text: str) -> Dict:
        """Извлечение сущностей из одного текста"""
        prompt = EXTRACTION_PROMPT.format(text=text)
        result = self.pipeline(prompt)
        
        if isinstance(result, list) and len(result) > 0:
            generated = result[0].get("generated_text", "")
        else:
            generated = str(result)
        
        try:
            json_start = generated.find("{")
            json_end = generated.rfind("}") + 1
            if json_start >= 0 and json_end > json_start:
                return json.loads(generated[json_start:json_end])
        except:
            pass
        
        return {"error": "Parse failed"}
    
    def extract_parallel(self, texts: List[str]) -> List[Dict]:
        """Параллельное извлечение сущностей"""
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            results = list(executor.map(self.extract_single, texts))
        return results
